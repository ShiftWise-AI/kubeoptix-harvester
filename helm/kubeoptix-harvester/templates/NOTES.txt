1. Install/upgrade the release:
   helm upgrade --install {{ .Release.Name }} . -n {{ include "kubeoptix-harvester.namespace" . }} --create-namespace

2. Check pods:
   oc get pods -n {{ include "kubeoptix-harvester.namespace" . }}

3. Check build resources:
   oc get is,bc -n {{ include "kubeoptix-harvester.namespace" . }}

4. If the Git repository is private, configure source secret:
   Option A: create secret manually and set .Values.build.sourceSecret.name
   Option B: set .Values.build.sourceSecret.create=true with username/token

5. Start image build (first deployment):
   oc start-build {{ include "kubeoptix-harvester.fullname" . }} -n {{ include "kubeoptix-harvester.namespace" . }} --follow

6. Check service:
   oc get svc {{ .Values.service.api.name }} -n {{ include "kubeoptix-harvester.namespace" . }}

7. Test API from cluster:
   oc run curl --rm -i --restart=Never -n {{ include "kubeoptix-harvester.namespace" . }} --image=curlimages/curl -- curl -sS http://{{ .Values.service.api.name }}:{{ .Values.service.api.port }}/health
